\begin{frame}{The idea: evaluate the nonlinear term off the mesh}
\begin{itemize}
\item \textbf{Before:} the reduced model's nonlinear integral $\int\psi\,u(u_x{+}u_y)$ was computed \emph{on the simulation mesh} (dense sum, fitted EQ, or a giant precomputed tensor). Cost and error both depended on the mesh.
\item \textbf{Hari's idea:} our bank $u(x)=G(x)c$ is a continuous function. Evaluate it and its \emph{exact} autodiff derivative at a few thousand textbook quadrature points (Gauss, lattice):
\[ N(c)=P^{\top}\big[(Bc)\odot(Dc)\big] \]
with $B,D,P$ precomputed once offline.
\item No fitting, no per-mesh retraining, and the same points at every mesh.
\end{itemize}
\end{frame}

\begin{frame}{3. Memory: wide banks become possible}
\fig{c1_3d_j4_memory_vs_Rp.png}
\say{The tensor needs 34\,GB at width 1024 and exceeds a whole H200 past about 1600. Off-mesh stays under 2\,GB.}
\end{frame}

\begin{frame}{4. Which quadrature rules work}
\fig{2d_rule_ladder.png}
\say{Gauss and lattices converge fast. Sobol needs about 4$\times$ more points; Smolyak fails. Deployed: Gauss (accurate setting), lattice (fast setting).}
\end{frame}

\begin{frame}{5. Why it works: the exact derivative}
\fig[0.5]{a2_gap_vs_h.png}
\say{The old error was the mesh's own differencing error: slope 1 for upwind, 2 for central, exactly as theory says. Mesh nodes with the exact derivative reproduce the off-mesh result, so the win is the exact derivative, and quadrature makes it cheap (about 14k points against millions of nodes).}
\end{frame}

\begin{frame}{6. Next bottleneck, time stepping: fixed with second-order schemes}
\fig{c2_error_vs_cost_acc.png}
\say{At equal cost, second-order schemes (CN-R, BDF2) have about 9$\times$ less error than backward Euler (circled).}
\end{frame}

\begin{frame}{7. Where we are: against new order-verified references}
\fig{s2_pareto2d.png}
\say{2D: median error 0.98 $\to$ 0.20\,\% at equal cost; about \textbf{14$\times$} faster than a full simulation that is second order in time. 3D: median 1.5 $\to$ 0.12\,\% at equal cost.}
\end{frame}

\begin{frame}{8. Honest limit: 3D speed}

\end{frame}
