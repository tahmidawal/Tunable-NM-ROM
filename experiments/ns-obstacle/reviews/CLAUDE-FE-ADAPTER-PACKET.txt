"""Small algebra for a training-only FE bridge; no NGSolve import or FE execution."""
from __future__ import annotations
import hashlib
import json
import numpy as np
from spaces import (canonicalize_eigenmodes, checked_cholesky, pressure_basis,
                    weighted_gram, weighted_pod)


def array_digest(*arrays):
    h = hashlib.sha256()
    for value in arrays:
        a = np.ascontiguousarray(value)
        h.update(str((a.shape, a.dtype.str)).encode())
        h.update(a.tobytes())
    return h.hexdigest()


def canonical_pod(coefficients, values, singular_values, weights):
    """Orient POD blocks using frozen FE coefficient order, keeping sampled fields exact."""
    c = np.asarray(coefficients, dtype=np.float64)
    v = np.asarray(values, dtype=np.float64)
    # Negative singular values are ascending. Cluster/sign rule is the existing rule.
    oriented, meta = canonicalize_eigenmodes(-np.asarray(singular_values), c,
                                            weighted_gram(v, weights))
    rotation, _, rank, _ = np.linalg.lstsq(c, oriented, rcond=1e-12)
    if rank != c.shape[1] or not np.allclose(c @ rotation, oriented, rtol=1e-10, atol=1e-12):
        raise ValueError("POD coefficient orientation lost rank")
    meta['rotation'] = rotation.tolist()
    return oriented, v @ rotation, meta


def coefficient_pod(velocity_coefficients, velocity_samples, pressure_coefficients,
                    pressure_samples, constant_coefficients, weights):
    """All inputs are dimensionless TRAINING corrections on one FE mesh.

    Coefficients [dof,case], velocity samples [case,point,component], pressure
    samples [case,point]. Returned basis coefficients preserve FE evaluation anywhere.
    """
    vc = np.asarray(velocity_coefficients, dtype=np.float64)
    pc = np.asarray(pressure_coefficients, dtype=np.float64)
    one = np.asarray(constant_coefficients, dtype=np.float64)
    u, um = weighted_pod(velocity_samples, weights, max_rank=8)
    cu = vc @ np.asarray(um['snapshot_combinations'])
    cu, u, orient_u = canonical_pod(cu, u, um['singular_values'][:u.shape[-1]], weights)
    um['eligible_velocity_prefixes'] = [r for r in (2, 4, 8) if r <= u.shape[-1]]
    if not um['eligible_velocity_prefixes']:
        raise ValueError('no registered velocity prefix passes the POD rank guard')
    p, pm = pressure_basis(pressure_samples, weights, 8)
    cp = one[:, None] / np.sqrt(np.sum(weights))
    if p.shape[-1] > 1:
        center_coeff = pc - one[:, None] * np.asarray(pm['snapshot_means'])[None, :]
        tail = center_coeff @ np.asarray(pm['centered_pod']['snapshot_combinations'])
        tail, p_tail, orient_p = canonical_pod(
            tail, p[:, 1:], pm['centered_pod']['singular_values'][:p.shape[-1]-1], weights)
        cp = np.c_[cp, tail]
        p = np.c_[p[:, 0], p_tail]
        pm['orientation'] = orient_p
    um['orientation'] = orient_u
    return {'velocity_coefficients': cu, 'pressure_coefficients': cp,
            'velocity_values': u, 'pressure_values': p,
            'metadata': {'velocity': um, 'pressure': pm}}


def reference_inf_sup(rhs, supremizers, pressure_gram):
    """Full-free-velocity H1 / SAME reduced pressure-L2 reference, not reduced admission."""
    gram = np.asarray(rhs).T @ np.asarray(supremizers)
    lp = checked_cholesky(pressure_gram)
    whitened = np.linalg.solve(lp, gram) @ np.linalg.inv(lp.T)
    if not np.allclose(whitened, whitened.T, rtol=1e-10, atol=1e-12):
        raise ValueError("supremizer reference Gram is not symmetric")
    eigenvalues = np.linalg.eigvalsh((whitened + whitened.T) / 2)
    if not np.isfinite(eigenvalues).all() or eigenvalues[0] <= 0:
        raise ValueError("full-velocity pressure coupling is singular")
    return {'beta': float(np.sqrt(eigenvalues[0])), 'squared_singular_values': eigenvalues.tolist(),
            'norms': 'dimensionless velocity H1 / pressure L2',
            'schur_gram': gram.tolist(), 'pressure_gram': np.asarray(pressure_gram).tolist()}


# Prospective fixed numerical-consistency gates, separate from physical error targets.
MODE_WEIGHTED_RELATIVE_TOLERANCE = 1e-10
MODE_GRAM_SPECTRAL_TOLERANCE = 1e-10


def verify_reconstructed_modes(actual, ideal, weights, label):
    """Compare actual FE evaluation of POD coefficients against their SVD sample modes.

    The relative discrepancy is checked per column in the original weighted L2 norm;
    the Gram spectral error is measured against identity. No new rank truncation or
    re-orthogonalization is allowed when either check fails. Return the ACTUAL Gram.
    """
    a = np.asarray(actual); target = np.asarray(ideal)
    if (a.shape != target.shape or a.ndim < 2 or a.shape[-1] < 1 or
        not np.isfinite(a).all() or not np.isfinite(target).all()):
        raise ValueError(f'{label}: invalid reconstructed/sample mode arrays')
    error_gram = weighted_gram(a-target, weights)
    ideal_gram = weighted_gram(target, weights)
    gram = weighted_gram(a, weights)
    ideal_norms = np.sqrt(np.diag(ideal_gram))
    if np.any(ideal_norms <= 0):
        raise ValueError(f'{label}: zero ideal mode')
    relative = np.sqrt(np.maximum(np.diag(error_gram), 0.))/ideal_norms
    gram_error = float(np.linalg.norm(gram-np.eye(a.shape[-1]), ord=2))
    report = {'field': label, 'mode_count': a.shape[-1],
              'weighted_relative_discrepancy_per_mode': relative.tolist(),
              'weighted_relative_discrepancy_max': float(np.max(relative)),
              'weighted_relative_tolerance': MODE_WEIGHTED_RELATIVE_TOLERANCE,
              'actual_gram_spectral_error': gram_error,
              'gram_spectral_tolerance': MODE_GRAM_SPECTRAL_TOLERANCE,
              'ideal_sample_gram_spectral_error': float(np.linalg.norm(ideal_gram-np.eye(a.shape[-1]), ord=2)),
              'actual_gram': gram.tolist(), 'rank_or_threshold_changed': False}
    report['passed'] = bool(np.max(relative) <= MODE_WEIGHTED_RELATIVE_TOLERANCE and
                            gram_error <= MODE_GRAM_SPECTRAL_TOLERANCE)
    if not report['passed']:
        raise ArithmeticError('FE coefficient mode reconstruction failed: '+json.dumps(report, allow_nan=False))
    return gram, report

def sha_file(path):
    h = hashlib.sha256()
    with Path(path).open('rb') as stream:
        for block in iter(lambda: stream.read(2**20), b''):
            h.update(block)
    return h.hexdigest()


def adapter_digest():

    return hashlib.sha256(json.dumps({name: sha_file(root/name) for name in
        ('fe_adapter.py', 'fe_algebra.py', 'spaces.py', 'weak.py', 'train_bank.py', 'MODEL-CONTRACT.json')}, sort_keys=True).encode()).hexdigest()

def load_npz_bounded(path):
    # Inspect ORIGINAL NPY headers before loading/converting any payload. This both
    # preserves f64 provenance and prevents f32->f64 conversion from hiding size growth.
    with zipfile.ZipFile(path) as z:
        entries = z.infolist()
        if sum(x.file_size for x in entries) > CAPS['archive_bytes']:
            raise MemoryError('uncompressed FE archive exceeds bound')
        if len({x.filename for x in entries}) != len(entries):
            raise ValueError('duplicate FE archive member')
        for entry in entries:
            if not entry.filename.endswith('.npy'):
                raise ValueError('FE archive contains a non-NPY member')
            with z.open(entry) as stream:
                version = np.lib.format.read_magic(stream)
                if version == (1, 0):
                    shape, _, dtype = np.lib.format.read_array_header_1_0(stream)
                elif version == (2, 0):
                    shape, _, dtype = np.lib.format.read_array_header_2_0(stream)
                else:
                    raise ValueError('unsupported NPY header version')
            if dtype.kind != 'f' or dtype.itemsize != 8:
                raise TypeError(f'FE array {entry.filename} must be stored float64, got {dtype}')
            check_allocation(shape)
    with np.load(path, allow_pickle=False) as z:
        result = {}
        for key in z.files:
            original = z[key]
            if original.dtype.kind != 'f' or original.dtype.itemsize != 8:
                raise TypeError(f'FE array {key} changed dtype after header verification')
            if not np.isfinite(original).all():
                raise ValueError(f'nonfinite FE array {key}')
            # Only byte order/layout may change here; original precision is already f64.
            result[key] = np.array(original, dtype=np.float64, copy=True)
    return result


def strict_scalar(value, label='FE scalar'):
    """Finite real scalar or singleton tuple/array; never select from a larger CF."""
    array = np.asarray(value)
    if array.size != 1 or array.dtype.kind not in 'fiu':
        raise ValueError(f'{label}: expected one real scalar, got shape {array.shape}, dtype {array.dtype}')
    scalar = float(array.reshape(-1)[0])
    if not math.isfinite(scalar):
        raise ValueError(f'{label}: nonfinite value')
    return scalar


def check_allocation(shape):
    count = math.prod(shape) * 8
    if count > CAPS['single_array_bytes']:
        raise MemoryError(f'proposed array exceeds adapter cap: {shape}')
    if resource.getrusage(resource.RUSAGE_SELF).ru_maxrss * 1024 + count > CAPS['process_rss_bytes']:
        raise MemoryError('adapter RSS plus proposed array exceeds cap')


def relative_gap(actual, expected, scale=1.):
    a = np.asarray(actual); b = np.asarray(expected)
    if a.shape != b.shape or not np.isfinite(a).all() or not np.isfinite(b).all():
        raise ValueError('round-trip shape/nonfinite mismatch')
    return float(np.max(np.abs(a-b), initial=0) / max(float(np.max(np.abs(b), initial=0)), scale))


def layout_record(mesh, velocity_space, pressure_space, ng):
    """Canonical JSON digest includes ordered VOL elements and local FE DOF numbers."""
    record = {'schema': 'ns-obstacle-fe-layout-v1', 'dimension': int(mesh.dim),
              'velocity_ndof': int(velocity_space.ndof), 'pressure_ndof': int(pressure_space.ndof),
              'velocity_free': [int(bool(v)) for v in velocity_space.FreeDofs()],
              'pressure_free': [int(bool(v)) for v in pressure_space.FreeDofs()], 'elements': []}
    for el in mesh.Elements(ng.VOL):
        record['elements'].append({'number': int(el.nr), 'type': str(el.type),
            'velocity_dofs': [int(i) for i in velocity_space.GetDofNrs(el)],
            'pressure_dofs': [int(i) for i in pressure_space.GetDofNrs(el)]})
    digest = hashlib.sha256(json.dumps(record, sort_keys=True, separators=(',', ':')).encode()).hexdigest()
    return record, digest


def _sparse(form, size):
    from scipy.sparse import coo_matrix
    if form.mat.nze > CAPS['matrix_nnz']:
        raise MemoryError('sparse nonzero bound')
    rows, cols, vals = form.mat.COO()
    matrix = coo_matrix((np.asarray(vals), (np.asarray(rows), np.asarray(cols))),
                        shape=(size, size)).tocsc()
    matrix.sum_duplicates(); matrix.sort_indices()
    gap = matrix-matrix.T
    if not np.isfinite(matrix.data).all() or (gap.nnz and np.max(np.abs(gap.data)) > 1e-12*max(np.max(np.abs(matrix.data)), 1e-300)):
        raise ArithmeticError('assembled FE metric is not finite symmetric')
    return matrix


def sparse_digest(matrix):
    a = matrix.tocsc(copy=True); a.sum_duplicates(); a.sort_indices()
    return array_digest(np.asarray(a.shape, dtype=np.int64), a.indptr, a.indices, a.data)


def bounded_factor(matrix):
    from scipy.sparse.linalg import splu
    # Same conservative fill model as FOM; a prospective guard, not a memory theorem.
    estimate = matrix.nnz*16*100 + matrix.shape[0]*8*32
    if (matrix.nnz > CAPS['matrix_nnz'] or estimate > CAPS['factor_estimate_bytes'] or
        resource.getrusage(resource.RUSAGE_SELF).ru_maxrss*1024 + estimate > CAPS['process_rss_bytes']):
        raise MemoryError('FE adapter factorization estimate exceeds fixed cap')
    started = time.perf_counter()
    factor = splu(matrix, permc_spec='COLAMD')
    return factor, {'estimate_bytes': int(estimate), 'seconds': time.perf_counter()-started,
                    'matrix_sha256': sparse_digest(matrix), 'backend': 'scipy SuperLU COLAMD'}


def bounded_eigenmodes(h1, mass, solve_h1, count):
    """H1 = M+K; generalized eigenmodes also diagonalize scalar Laplace K.

    One look-ahead eigenpair protects the truncation from splitting a degenerate block.
    No retry, larger candidate bank, or dense fallback is selected from field errors.
    """
    from scipy.sparse.linalg import eigsh, LinearOperator
    n = h1.shape[0]
    if n <= count+2:
        raise ValueError('not enough free DOFs for frozen eigenbank plus look-ahead')
    v0 = np.random.default_rng(27102026).normal(size=n)
    inv = LinearOperator(h1.shape, matvec=solve_h1, dtype=np.float64)
    lam, vec = eigsh(h1, k=count+1, M=mass, sigma=0., which='LM', OPinv=inv,
                    v0=v0, tol=1e-12, maxiter=CAPS['eigensolver_iterations'],
                    ncv=min(n-1, 2*(count+1)+1))
    idx = np.argsort(lam, kind='stable'); lam = lam[idx]; vec = vec[:, idx]
    if abs(lam[count]-lam[count-1]) <= 1e-10*max(1., abs(lam[count])):
        raise ValueError('candidate cap splits eigenvalue cluster; stop for review')
    look_ahead = float(lam[count]); boundary_gap = float(lam[count]-lam[count-1])
    lam = lam[:count]; vec = vec[:, :count]
    av = h1 @ vec; mv = mass @ vec
    errors = np.linalg.norm(av-mv*lam, axis=0) / np.maximum(
        np.linalg.norm(av, axis=0)+np.abs(lam)*np.linalg.norm(mv, axis=0), 1e-300)
    if np.max(errors) > 1e-9:
        raise ArithmeticError('generalized eigenpair residual gate')
    vec, orientation = canonicalize_eigenmodes(lam, vec, vec.T @ mv)
    av = h1 @ vec; mv = mass @ vec
    oriented_errors = np.linalg.norm(av-mv*lam, axis=0) / np.maximum(
        np.linalg.norm(av, axis=0)+np.abs(lam)*np.linalg.norm(mv, axis=0), 1e-300)
    if np.max(oriented_errors) > 1e-9:
        raise ArithmeticError('oriented eigenpair residual gate')
    return vec, {'eigenvalues_H1': lam.tolist(), 'eigenvalues_Laplace': (lam-1).tolist(),
                 'residuals': errors.tolist(), 'oriented_residuals': oriented_errors.tolist(),
                 'orientation': orientation,
                 'h1_sha256': sparse_digest(h1), 'mass_sha256': sparse_digest(mass),
                 'eigensolver': 'scipy eigsh shift-invert sigma0 tol1e-12 seed27102026',
                 'look_ahead_cluster_complete': True, 'look_ahead_eigenvalue': look_ahead,
                 'boundary_eigenvalue_gap': boundary_gap}

        self.Q = ng.H1(self.mesh, order=self.cfg['pressure_order'])

        if self.X.ndof > CAPS['mixed_dofs']:
            raise MemoryError('mixed DOF cap before any matrix assembly')

    def mapped_rule(self, boundary=False):
        ng = self.ng; vb = ng.BND if boundary else ng.VOL
        cap = CAPS['boundary_points'] if boundary else CAPS['volume_points']
        points = []; weights = []; labels = []; normals = []; jacobians = []
        normal = ng.specialcf.normal(2)
        for el in self.mesh.Elements(vb):
            tr = self.mesh.GetTrafo(el)
            for ip in ng.IntegrationRule(el.type, self.cfg['diagnostic_order']):
                if len(points) >= cap:
                    raise MemoryError('mapped integration point cap')
                mip = tr(ip); w = strict_scalar(ip.weight, 'reference weight')*strict_scalar(mip.measure, 'mapped measure')
                if not np.isfinite(w) or w <= 0:
                    raise ValueError('nonpositive mapped integration weight')
                points.append(np.array(mip.point, copy=True)); weights.append(w)
                if boundary:
                    labels.append(str(el.mat)); normals.append(np.array(normal(mip), copy=True))
                else:
                    jac = np.array(mip.jacobi, copy=True)
                    if np.linalg.det(jac) <= 0:
                        raise ValueError('nonpositive mapped volume Jacobian')
                    jacobians.append(jac)
        result = {'points': np.asarray(points), 'weights': np.asarray(weights)}
        if boundary:
            result.update(labels=np.asarray(labels), normals_fluid=np.asarray(normals))
        else:
            result['jacobians'] = np.asarray(jacobians)
        return result

    def evaluate(self, coefficients, vector=True, boundary=False, points=None):
        """FE coefficients -> dimensionless values and physical-coordinate derivatives.

        Requested raw points are never nudged to the FE mesh. Boundary derivatives
        use BoundaryFromVolumeCF only for the mapped BND rule, not for normal vectors.
        """
        ng = self.ng; c = np.asarray(coefficients, dtype=np.float64)
        space = self.V if vector else self.Q
        if c.ndim == 1: c = c[:, None]
        if c.shape[0] != space.ndof or not np.isfinite(c).all():
            raise ValueError('invalid FE coefficient matrix')
        if points is not None and boundary:
            raise ValueError('raw points use volume lookup; mapped boundary must retain BND transform')
        if points is None:
            n = len(self.boundary['points'] if boundary else self.volume['points'])
        else:
            points = np.asarray(points, dtype=np.float64)
            if points.ndim != 2 or points.shape[1] != 2 or not np.isfinite(points).all():
                raise ValueError('finite 2D probe coordinates required')
            if len(points) > CAPS['volume_points']: raise MemoryError('probe cap')
            n = len(points)
        shape = (n, 2, c.shape[1]) if vector else (n, c.shape[1])
        grad_shape = (n, 2, 2, c.shape[1]) if vector else (n, 2, c.shape[1])
        check_allocation(shape); check_allocation(grad_shape)
        values = np.empty(shape); gradients = np.empty(grad_shape)
        for j in range(c.shape[1]):
            g = ng.GridFunction(space); g.vec.FV().NumPy()[:] = c[:, j]
            derivative = ng.Grad(g)
            if boundary: derivative = ng.BoundaryFromVolumeCF(derivative)
            for i, mip in enumerate(self.iter_mapped(boundary, points)):
                values[..., j][i] = (np.asarray(g(mip)) if vector else strict_scalar(g(mip), 'FE scalar point value'))
                gradients[..., j][i] = np.asarray(derivative(mip)).reshape((2, 2) if vector else (2,))
        if not np.isfinite(values).all() or not np.isfinite(gradients).all():
            raise ValueError('nonfinite reconstructed FE fields')
        return values, gradients

    def iter_mapped(self, boundary=False, points=None):
        # Evaluate while each element transformation remains alive. Do not retain
        # mapped C++ points after their transformation leaves scope.
        if points is not None:
            for point in points:
                mip = self.mesh(*point)
                if mip.nr < 0: raise ValueError('probe outside saved FE geometry; no nudge')
                yield mip
            return
        ng = self.ng
        for el in self.mesh.Elements(ng.BND if boundary else ng.VOL):
            tr = self.mesh.GetTrafo(el)
            for ip in ng.IntegrationRule(el.type, self.cfg['diagnostic_order']):
                yield tr(ip)

    def _load_coefficients(self):
        ng = self.ng; solutions = []; stokes = []
        for c in self.cases:
            data = load_npz_bounded(c['folder']/'full_rule.npz')
            checks = {}
            for key in ('points', 'weights', 'jacobians'):
                if key not in data:
                    raise ValueError(f'future G2 export must add {key}; G1 archive is insufficient')
                checks[key] = relative_gap(self.volume[key], data[key], self.D if key == 'points' else 1.)
            saved = ng.GridFunction(self.X); saved.Load(str(c['folder']/'solution.sol'))
            solution = data['solution_coefficients']; lift = data['stokes_coefficients']
            if solution.shape != (self.X.ndof,) or lift.shape != solution.shape:
                raise ValueError('coefficient block layout mismatch')
            checks['sol_vs_coefficients'] = relative_gap(saved.vec.FV().NumPy(), solution)
            uv, gu = self.evaluate(solution[:self.V.ndof]); pp, _ = self.evaluate(solution[self.V.ndof:], False)
            for key, actual, scale in [('velocity', uv[..., 0], self.U),
                ('velocity_gradient', gu[..., 0], self.U/self.D), ('pressure', pp[:, 0], self.U**2)]:
                checks[key] = relative_gap(actual, data[key], scale)
            b = json.loads((c['folder']/'boundary_rule.json').read_text())
            for key, actual in [('point', self.boundary['points']), ('weight', self.boundary['weights']),
                                ('normal', self.boundary['normals_fluid'])]:
                checks['boundary_'+key] = relative_gap(actual, np.asarray([r[key] for r in b]))
            if [r['label'] for r in b] != self.boundary['labels'].tolist():
                raise ValueError('boundary enumeration differs on reload')
            bv, bg = self.evaluate(solution[:self.V.ndof], boundary=True)
            bp, _ = self.evaluate(solution[self.V.ndof:], False, boundary=True)
            for key, actual, scale in [('velocity', bv[..., 0], self.U),
                ('velocity_gradient', bg[..., 0], self.U/self.D), ('pressure', bp[:, 0], self.U**2)]:
                checks['boundary_'+key] = relative_gap(actual, [r[key] for r in b], scale)

"""Mass-weighted POD/ordering, fixed Gram whitening, and explicit numerical rank gates."""
from __future__ import annotations
import numpy as np

RANK_CUTOFF=1e-12


def positive_weights(weights,n):
    w=np.asarray(weights,dtype=np.float64)
    if w.shape != (n,) or not np.isfinite(w).all() or np.any(w<=0):
        raise ValueError("finite strictly positive physical-volume weights required")
    return w


def checked_cholesky(gram):
    g=np.asarray(gram,dtype=np.float64)
    if g.ndim!=2 or g.shape[0]!=g.shape[1] or not np.isfinite(g).all():
        raise ValueError("finite square Gram required")
    if np.linalg.norm(g-g.T)>1e-12*max(np.linalg.norm(g),1e-300):
        raise ValueError("Gram is not symmetric")
    eig=np.linalg.eigvalsh(g)
    if eig[-1]<=0 or eig[0]<=RANK_CUTOFF*eig[-1]:
        raise ValueError("Gram fails fixed 1e-12 eigenvalue rank gate; no hidden ridge")
    return np.linalg.cholesky(g)


def weighted_gram(values,weights):
    a=np.asarray(values,dtype=np.float64)
    w=positive_weights(weights,len(a))
    flat=a.reshape(-1,a.shape[-1])
    repeated=np.repeat(w,int(np.prod(a.shape[1:-1])))
    return flat.T@(repeated[:,None]*flat)


def ordered_transform(bank_values,correction_snapshots,weights):
    """QR+training SVD; snapshots[s,n,d], bank[n,d,r]. No validation access."""
    g=np.asarray(bank_values,dtype=np.float64)
    u=np.asarray(correction_snapshots,dtype=np.float64)
    if u.ndim!=3 or u.shape[1:]!=g.shape[:2] or len(u)<g.shape[-1]:
        raise ValueError("insufficient snapshots or incompatible shape")
    sw=np.repeat(np.sqrt(positive_weights(weights,len(g))),g.shape[1])
    a=g.reshape(-1,g.shape[-1])*sw[:,None]
    singular=np.linalg.svd(a,compute_uv=False)
    if singular[-1]<=RANK_CUTOFF*singular[0]:
        raise ValueError("neural columns rank deficient before ordering")
    q,rf=np.linalg.qr(a,mode="reduced")
    coefficients=(u.reshape(len(u),-1)*sw[None,:])@q
    _,importance,vt=np.linalg.svd(coefficients,full_matrices=False)
    rotation=vt.T
    sign=np.sign(rotation[np.argmax(np.abs(rotation),axis=0),np.arange(rotation.shape[1])])
    rotation*=np.where(sign==0,1,sign)
    transform=np.linalg.solve(rf,rotation)
    return transform,{"basis_singular_values":singular.tolist(),"training_importance":importance.tolist(),
                      "rank_cutoff":RANK_CUTOFF,"training_count":len(u),"mean_centered":False}


def weighted_pod(snapshots,weights,max_rank=None):
    """Snapshots[s,n,...]; no automatic mean or pressure-gauge shift."""
    u=np.asarray(snapshots,dtype=np.float64)
    if u.ndim<2 or not np.isfinite(u).all():
        raise ValueError("finite snapshots required")
    n=u.shape[1]
    components=int(np.prod(u.shape[2:]))
    sw=np.repeat(np.sqrt(positive_weights(weights,n)),components)
    left,sv,vt=np.linalg.svd((u.reshape(len(u),-1)*sw).T,full_matrices=False)
    numerical_rank=int(np.sum(sv>RANK_CUTOFF*sv[0])) if len(sv) and sv[0]>0 else 0
    rank=numerical_rank if max_rank is None else min(int(max_rank),numerical_rank)
    if rank<1:
        raise ValueError("no nonzero POD mode")
    basis=(left[:,:rank]/sw[:,None]).reshape(*u.shape[1:],rank)
    return basis,{"singular_values":sv.tolist(),"original_count":len(sv),"numerical_rank":numerical_rank,
                  "retained_rank":rank,"rank_cutoff":RANK_CUTOFF,"mean_centered":False,
                  "snapshot_combinations":(vt[:rank].T/sv[:rank]).tolist()}


def pressure_basis(training_pressure_corrections,weights,velocity_prefix):
    """Constant first; then mass POD of centered TRAINING pressure corrections.

    This changes basis construction, not the physical pressure convention: reconstructed
    pressure keeps its solved constant coefficient. No gauge constraint is imposed.
    """
    p=np.asarray(training_pressure_corrections,dtype=np.float64)
    if p.ndim!=2 or not np.isfinite(p).all() or velocity_prefix not in (2,4,8):
        raise ValueError("finite training pressure snapshots and declared prefix required")
    w=positive_weights(weights,p.shape[1]);volume=w.sum()
    means=p@w/volume
    centered=p-means[:,None]
    constant=np.ones(p.shape[1])/np.sqrt(volume)
    matrix=(centered*np.sqrt(w)).T
    left,sv,vt=np.linalg.svd(matrix,full_matrices=False)
    reference=max(1.,float(np.linalg.svd((p*np.sqrt(w)).T,compute_uv=False)[0]))
    numerical_rank=int(np.sum(sv>RANK_CUTOFF*reference))
    if numerical_rank==0:
        return constant[:,None],{"pressure_rank":1,"centered_numerical_rank":0,"constant_first":True,
                                 "physical_gauge_shift":False,"snapshot_means":means.tolist(),
                                 "rank_reference":reference,"centered_singular_values":sv.tolist()}
    rank=min(velocity_prefix-1,numerical_rank)
    modes=left[:,:rank]/np.sqrt(w[:,None])
    meta={"numerical_rank":numerical_rank,"retained_rank":rank,"singular_values":sv.tolist(),
          "rank_reference":reference,"rank_reference_definition":"max(1,original weighted pressure snapshot sigma_max)",
          "relative_rank_cutoff":RANK_CUTOFF,"snapshot_combinations":(vt[:rank].T/sv[:rank]).tolist()}
    result=np.c_[constant,modes]
    return result,{"pressure_rank":result.shape[1],"centered_numerical_rank":meta["numerical_rank"],
                   "constant_first":True,"physical_gauge_shift":False,"snapshot_means":means.tolist(),
                   "centered_pod":meta}


def canonicalize_eigenmodes(eigenvalues,eigenvectors,metric_gram,degeneracy_relative=1e-10):
    """Deterministic orientation inside eigenvalue clusters using ordered FE DOF values.

    Input eigenvectors are columns, already orthonormal in the stated FE metric. Largest
    residual DOF-row norm is the pivot; np.argmax fixes ties to smallest FE DOF index.
    Canonical result is invariant under orthogonal rotations of a degenerate block.
    """
    lam=np.asarray(eigenvalues,dtype=np.float64);v=np.asarray(eigenvectors,dtype=np.float64)
    if v.ndim!=2 or v.shape[1]!=len(lam) or np.any(np.diff(lam)<0):
        raise ValueError("sorted eigenpairs required")
    if not np.allclose(metric_gram,np.eye(len(lam)),atol=1e-10,rtol=1e-10):
        raise ValueError("eigenvectors must be metric-orthonormal before canonicalization")
    out=np.empty_like(v);groups=[];start=0
    while start<len(lam):
        end=start+1
        while end<len(lam) and abs(lam[end]-lam[start])<=degeneracy_relative*max(1.,abs(lam[start]),abs(lam[end])):
            end+=1
        rows=v[:,start:end].copy();rotation=[];pivots=[]
        for _ in range(end-start):
            pivot=int(np.argmax(np.sum(rows*rows,axis=1)))
            direction=rows[pivot].copy();norm=np.linalg.norm(direction)
            if norm<=1e-14:
                raise ValueError("degenerate-block orientation lost rank")
            direction/=norm
            rotation.append(direction);pivots.append(pivot)
            rows-=np.outer(rows@direction,direction)
        out[:,start:end]=v[:,start:end]@np.column_stack(rotation)
        groups.append({"start":start,"end":end,"pivot_FE_dofs":pivots})
        start=end
    return out,{"groups":groups,"relative_eigenvalue_cluster_tolerance":degeneracy_relative,
                "tie_rule":"largest residual DOF-row norm, first FE DOF index on exact tie"}


def select_test_columns(values,weights,target_count,gradients=None):
    """Ordered candidates -> fixed full-rule H1 or L2 independent normalized test bank.

    Momentum candidates: all maximum pressure-prefix supremizers, then canonical vector
    H1 eigenmodes. Continuity: constant, pressure modes, then canonical scalar Laplace
    modes. This routine receives that immutable enumeration and never sees state errors.
    """
    v=np.asarray(values,dtype=np.float64);w=positive_weights(weights,len(v))
    flat=v.reshape(-1,v.shape[-1])*np.repeat(np.sqrt(w),int(np.prod(v.shape[1:-1])))[:,None]
    if gradients is not None:
        g=np.asarray(gradients,dtype=np.float64)
        gf=g.reshape(-1,g.shape[-1])*np.repeat(np.sqrt(w),int(np.prod(g.shape[1:-1])))[:,None]
        flat=np.r_[flat,gf]
    selected=[];rejected=[]
    for j in range(flat.shape[1]):
        sv=np.linalg.svd(flat[:,selected+[j]],compute_uv=False)
        if not len(sv) or sv[0]==0 or sv[-1]<=RANK_CUTOFF*sv[0]:
            rejected.append(j);continue
        selected.append(j)
        if len(selected)==target_count:
            break
    if len(selected)!=target_count:
        raise ValueError(f"test enumeration has {len(selected)} independent candidates, needs {target_count}")
    _,r=np.linalg.qr(flat[:,selected],mode="reduced")
    transform=np.linalg.inv(r)
    return v[...,selected]@transform, (None if gradients is None else g[...,selected]@transform), {
        "candidate_indices":selected,"rejected_indices":rejected,"target_count":target_count,
        "transform":transform.tolist(),"rank_cutoff":RANK_CUTOFF,
        "metric":"dimensionless H1" if gradients is not None else "dimensionless L2"}


def projection_floor(basis,snapshots,weights):

    u=np.asarray(snapshots,dtype=np.float64)

def inf_sup(divergence,velocity_gram,pressure_gram,reference_beta):
    """Standard velocity H1/pressure L2 coupling; reference MUST use those same norms.

    The coordinator corrected draft A1's inf-sup norm before numerical execution.
    Jacobian unknown whitening separately continues to use velocity/pressure L2.
    """
    lv=checked_cholesky(velocity_gram)
    lp=checked_cholesky(pressure_gram)
    b=np.asarray(divergence,dtype=np.float64)
    if b.shape!=(len(lp),len(lv)) or len(lp)>len(lv):
        raise ValueError("pressure divergence coupling shape or dimension invalid")
    scaled=np.linalg.solve(lp,b)@np.linalg.inv(lv.T)
    sv=np.linalg.svd(scaled,compute_uv=False)
    beta=float(sv[-1])
    if not np.isfinite(reference_beta) or reference_beta<=0:
        raise ValueError("positive full-velocity/same-pressure reference required")
    ratio=beta/reference_beta
    return {"beta":beta,"reference_beta":float(reference_beta),"ratio":ratio,
            "singular_values":sv.tolist(),"norms":"dimensionless velocity H1, pressure L2",
            "passed":bool(beta>=1e-3 and ratio>=0.5)}

"""Gradient-form steady mixed weak operators in FIXED dimensionless units.

Array axes: point, vector component, derivative coordinate, column. Volume points use
x/D, velocities u/U, pressure p/U**2, nu_hat=nu/(U*D). Pressure lift is nu_hat*p_lift.

@dataclass
class WeakRule:

    lift_gradient: np.ndarray
    pressure_lift_per_nu: np.ndarray
    velocity: np.ndarray

    test_gradient: np.ndarray
    scalar_tests: np.ndarray

    def __post_init__(self):

            setattr(self,name,np.asarray(getattr(self,name),dtype=np.float64))
        n,d=self.lift.shape
        nv=self.velocity.shape[-1]
        m=self.tests.shape[-1]
        shapes={"lift_gradient":(n,d,d),"pressure_lift_per_nu":(n,),
                "velocity":(n,d,nv),"velocity_gradient":(n,d,d,nv),

                raise ValueError(f"bad {name} shape")

            raise ValueError("nonfinite rule arrays")

    def direct(self,state,nu_hat):

        nv=self.velocity.shape[-1]

            raise ValueError("state dimension mismatch")

        adv=np.einsum("nj,nij->ni",u,du)
        divt=np.einsum("niim->nm",self.test_gradient)
        rm=(np.einsum("n,ni,nim->m",self.weights,adv,self.tests)+
            nu_hat*np.einsum("n,nij,nijm->m",self.weights,du,self.test_gradient)-
            np.einsum("n,n,nm->m",self.weights,p,divt))
        rc=np.einsum("n,nq,n->q",self.weights,self.scalar_tests,np.trace(du,axis1=1,axis2=2))
        return rm,rc

    def tensor(self,chunk=1024,formulation="weak_residual_least_squares"):
        if formulation=="mixed_Galerkin" and not (
                np.array_equal(self.tests,self.velocity) and
                np.array_equal(self.test_gradient,self.velocity_gradient) and
                np.array_equal(self.scalar_tests,self.pressure)):
            raise ValueError("Galerkin tests must equal the actual velocity/pressure trial modes")
        nv=self.velocity.shape[-1]
        ext=np.concatenate((self.lift[:,:,None],self.velocity),axis=2)
        dext=np.concatenate((self.lift_gradient[:,:,:,None],self.velocity_gradient),axis=3)
        m=self.tests.shape[-1]
        tensor=np.zeros((m,nv+1,nv+1),dtype=np.float64)
        for start in range(0,len(ext),chunk):
            s=slice(start,min(start+chunk,len(ext)))
            # Separate chunks avoid materializing point*test*trial*trial arrays.
            tensor+=np.einsum("n,nim,nja,nijb->mab",self.weights[s],self.tests[s],ext[s],dext[s],optimize=True)
        viscous=np.einsum("n,nija,nijm->ma",self.weights,dext,self.test_gradient,optimize=True)
        divt=np.einsum("niim->nm",self.test_gradient)
        pressure=-np.einsum("n,nk,nm->mk",self.weights,self.pressure,divt)
        plift=-np.einsum("n,n,nm->m",self.weights,self.pressure_lift_per_nu,divt)
        continuity=np.einsum("n,nq,niia->qa",self.weights,self.scalar_tests,dext)
        vg=(np.einsum("n,nim,nil->ml",self.weights,self.tests,self.tests)+
            np.einsum("n,nijm,nijl->ml",self.weights,self.test_gradient,self.test_gradient))
        qg=np.einsum("n,nq,nr->qr",self.weights,self.scalar_tests,self.scalar_tests)
        ug=np.einsum("n,nia,nib->ab",self.weights,self.velocity,self.velocity)
        pg=np.einsum("n,na,nb->ab",self.weights,self.pressure,self.pressure)
        return WeakOperator(tensor,viscous,pressure,plift,continuity,vg,qg,ug,pg,formulation)


@dataclass
class WeakOperator:
    convection: np.ndarray
    viscous: np.ndarray
    pressure: np.ndarray
    pressure_lift: np.ndarray
    continuity: np.ndarray
    velocity_test_gram: np.ndarray
    scalar_test_gram: np.ndarray
    velocity_unknown_gram: np.ndarray
    pressure_unknown_gram: np.ndarray
    formulation: str = "weak_residual_least_squares"

    def __post_init__(self):
        for name in self.__dataclass_fields__:
            if name!="formulation":
                setattr(self,name,np.asarray(getattr(self,name),dtype=np.float64))
        self.lv=checked_cholesky(self.velocity_test_gram)
        self.lq=checked_cholesky(self.scalar_test_gram)
        a=checked_cholesky(self.velocity_unknown_gram)
        p=checked_cholesky(self.pressure_unknown_gram)
        self.unknown_cholesky=np.zeros((len(a)+len(p),len(a)+len(p)))
        self.unknown_cholesky[:len(a),:len(a)]=a
        self.unknown_cholesky[len(a):,len(a):]=p
        self.unknown_map=np.linalg.inv(self.unknown_cholesky.T)
        self.nv=len(a)
        self.np=len(p)
        if self.formulation=="weak_residual_least_squares":
            if len(self.lv)+len(self.lq)<=self.nv+self.np:
                raise ValueError("weak LS requires strictly more independent test rows than unknowns")
        elif self.formulation=="mixed_Galerkin":
            if (len(self.lv),len(self.lq))!=(self.nv,self.np):
                raise ValueError("mixed Galerkin needs matched velocity and pressure test dimensions")
        else:
            raise ValueError("unsupported reduced formulation")

    def terms(self,state,nu_hat):
        z=np.asarray(state,dtype=np.float64)
        if z.shape!=(self.nv+self.np,) or not np.isfinite(z).all() or not np.isfinite(nu_hat) or nu_hat<=0:
            raise ValueError("finite state and positive viscosity required")
        x=np.r_[1.,z[:self.nv]]
        return {"advection":np.einsum("mab,a,b->m",self.convection,x,x),
                "viscous":nu_hat*self.viscous@x,
                "pressure":self.pressure@z[self.nv:],
                "pressure_lifting":nu_hat*self.pressure_lift,
                "continuity":self.continuity@x}

    def residual(self,state,nu_hat,whiten=True):
        terms=self.terms(state,nu_hat)
        m=sum(terms[k] for k in ("advection","viscous","pressure","pressure_lifting"))
        c=terms["continuity"]
        if whiten:
            m=np.linalg.solve(self.lv,m)
            c=np.linalg.solve(self.lq,c)
        return np.r_[m,c]

    def jacobian_terms(self,state,nu_hat):
        z=np.asarray(state,dtype=np.float64)
        x=np.r_[1.,z[:self.nv]]
        sym=self.convection+self.convection.transpose(0,2,1)
        shape=(len(self.lv)+len(self.lq),self.nv+self.np)
        result={k:np.zeros(shape) for k in ("advection","viscous","pressure","continuity")}
        result["advection"][:len(self.lv),:self.nv]=np.einsum("mab,b->ma",sym,x)[:,1:]
        result["viscous"][:len(self.lv),:self.nv]=nu_hat*self.viscous[:,1:]
        result["pressure"][:len(self.lv),self.nv:]=self.pressure
        result["continuity"][len(self.lv):,:self.nv]=self.continuity[:,1:]
        return result

    def whiten_matrix(self,jac,columns=False):
        out=np.r_[np.linalg.solve(self.lv,jac[:len(self.lv)]),np.linalg.solve(self.lq,jac[len(self.lv):])]
        return out@self.unknown_map if columns else out

    def jacobian(self,state,nu_hat,whiten=True,columns=False):
        jac=sum(self.jacobian_terms(state,nu_hat).values())
        if whiten:
            return self.whiten_matrix(jac,columns)
        if columns:
            raise ValueError("column-only whitening is not a supported contract")
        return jac

    def condition(self,state,nu_hat):
        sv=np.linalg.svd(self.jacobian(state,nu_hat,columns=True),compute_uv=False)
        rank=int(np.sum(sv>1e-12*sv[0])) if len(sv) and sv[0]>0 else 0
        cond=float(sv[0]/sv[-1]) if rank==len(sv) else float("inf")
        return {"rank":rank,"columns":len(sv),"singular_values":sv.tolist(),"condition":cond,
                "passed":bool(rank==self.nv+self.np and cond<=1e10)}

    def quadrature_gap(self,other,state,nu_hat):
        """Full-rule normalization and whitening stay frozen; compare another raw operator."""
        base=self.terms(state,nu_hat)
        diff=self.residual(state,nu_hat,False)-other.residual(state,nu_hat,False)
        wr=np.r_[np.linalg.solve(self.lv,diff[:len(self.lv)]),np.linalg.solve(self.lq,diff[len(self.lv):])]
        denominator=sum(np.linalg.norm(np.linalg.solve(self.lv,base[k])) for k in
                        ("advection","viscous","pressure","pressure_lifting"))
        denominator+=np.linalg.norm(np.linalg.solve(self.lq,base["continuity"]))
        denominator=max(float(denominator),1e-12)
        jd=self.jacobian(state,nu_hat,False)-other.jacobian(state,nu_hat,False)
        jnum=float(np.linalg.norm(self.whiten_matrix(jd,columns=True)))
        jden=max(sum(np.linalg.norm(self.whiten_matrix(v,columns=True)) for v in
                     self.jacobian_terms(state,nu_hat).values()),1e-12)
        return {"operator_numerator":float(np.linalg.norm(wr)),"operator_denominator":denominator,
                "operator_relative":float(np.linalg.norm(wr))/denominator,
                "jacobian_numerator":jnum,"jacobian_denominator":float(jden),"jacobian_relative":jnum/jden}


def physical_to_dimensionless(weights,lift,lift_gradient,pressure_lift_per_nu,D,U,dimension):
    """FE exporter bridge. Test values are dimensionless; test gradients separately *D."""
    if not D>0 or not U>0 or dimension not in (2,3):
        raise ValueError("positive physical scales and dimension 2/3 required")
    return {"weights":np.asarray(weights)/D**dimension,"lift":np.asarray(lift)/U,
            "lift_gradient":np.asarray(lift_gradient)*D/U,
            "pressure_lift_per_nu":np.asarray(pressure_lift_per_nu)*D/U}


def force_moments(boundary_weights,n_body,velocity_gradient,pressure_basis,lift_gradient,pressure_lift_per_nu,reference_area):
    """Dimensionless force coefficient C=2*integral(sigma*n_body)/area, separately linear.

    n_body points from solid into fluid. In 2D reference_area=1 after scaling by D;
    in 3D sphere reference_area=pi/4. Geometry mapping/admission is the caller's gate.
    """
    w=positive_weights(boundary_weights,len(n_body))
    n=np.asarray(n_body,dtype=np.float64)
    if not np.allclose(np.linalg.norm(n,axis=1),1,rtol=1e-10,atol=1e-12) or reference_area<=0:
        raise ValueError("unit body normals and positive reference area required")
    g=np.asarray(velocity_gradient)
    gl=np.asarray(lift_gradient)
    factor=2/reference_area
    return {"velocity_per_nu":factor*np.einsum("n,nijr,nj->ir",w,g+g.transpose(0,2,1,3),n),
            "pressure":-factor*np.einsum("n,nr,ni->ir",w,pressure_basis,n),
            "lifting_per_nu":factor*(np.einsum("n,nij,nj->i",w,gl+gl.transpose(0,2,1),n)-
                                      np.einsum("n,n,ni->i",w,pressure_lift_per_nu,n))}
